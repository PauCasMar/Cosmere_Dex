-- Crear la base de datos
CREATE DATABASE IF NOT EXISTS cosmere_db;
USE cosmere_db;

-- Tabla de mundos
CREATE TABLE mundos (
    id_mundo INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    descripcion TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Tabla de libros
CREATE TABLE libros (
    id_libro INT AUTO_INCREMENT PRIMARY KEY,
    titulo VARCHAR(200) NOT NULL,
    anio_publicacion YEAR NOT NULL,
    numero_paginas INT NOT NULL,
    id_mundo INT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_mundo) REFERENCES mundos(id_mundo) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Tabla de personajes
CREATE TABLE personajes (
    id_personaje INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(150) NOT NULL,
    id_libro INT NOT NULL,
    genero ENUM('hombre', 'mujer') NOT NULL,
    imagen_url VARCHAR(500),
    importancia ENUM('protagonista', 'personaje secundario', 'personaje no importante') NOT NULL,
    rol ENUM('villano', 'heroe', 'neutral') DEFAULT 'neutral',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_libro) REFERENCES libros(id_libro) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Tabla de habilidades
CREATE TABLE habilidades (
    id_habilidad INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    descripcion TEXT,
    tipo VARCHAR(100), -- Por ejemplo: Alomancia, Deshacimiento, Vinculación, etc.
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Tabla intermedia: relación muchos a muchos entre personajes y habilidades
CREATE TABLE personajes_habilidades (
    id_personaje INT NOT NULL,
    id_habilidad INT NOT NULL,
    PRIMARY KEY (id_personaje, id_habilidad),
    FOREIGN KEY (id_personaje) REFERENCES personajes(id_personaje) ON DELETE CASCADE,
    FOREIGN KEY (id_habilidad) REFERENCES habilidades(id_habilidad) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Tabla intermedia: relación muchos a muchos entre mundos y personajes
CREATE TABLE mundos_personajes (
    id_mundo INT NOT NULL,
    id_personaje INT NOT NULL,
    PRIMARY KEY (id_mundo, id_personaje),
    FOREIGN KEY (id_mundo) REFERENCES mundos(id_mundo) ON DELETE CASCADE,
    FOREIGN KEY (id_personaje) REFERENCES personajes(id_personaje) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Tabla intermedia: relación muchos a muchos entre libros y personajes
CREATE TABLE libros_personajes (
    id_libro INT NOT NULL,
    id_personaje INT NOT NULL,
    PRIMARY KEY (id_libro, id_personaje),
    FOREIGN KEY (id_libro) REFERENCES libros(id_libro) ON DELETE CASCADE,
    FOREIGN KEY (id_personaje) REFERENCES personajes(id_personaje) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Índices adicionales para mejorar el rendimiento
CREATE INDEX idx_libro_anio ON libros(anio_publicacion);
CREATE INDEX idx_personaje_nombre ON personajes(nombre);
CREATE INDEX idx_personaje_importancia ON personajes(importancia);
CREATE INDEX idx_mundo_nombre ON mundos(nombre);
